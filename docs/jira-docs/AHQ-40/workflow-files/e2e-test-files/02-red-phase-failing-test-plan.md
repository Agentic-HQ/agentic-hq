## Step 1: Create Shared Vitest Temp Git Dir Fixture

**Create** `tests/shared/fixtures.ts`

This is a `test.extend()` fixture providing a temp directory initialised as a git repo, with auto-cleanup. Placed in `tests/shared/` so it can be reused by any test type (unit, integration, smoke, e2e). Perplexity confirmed this works: fixture files are just TypeScript modules importable from anywhere, and `tests/shared/` is outside all test `include` globs so it won't be picked up as a test itself.

Pattern from [sdorra.dev](https://sdorra.dev/posts/2024-02-12-vitest-tmpdir) and agreed in the AI summary.

```typescript
/**
 * Shared Vitest Test Fixtures
 *
 * Uses Vitest's test.extend() to provide reusable test fixtures with automatic
 * setup and teardown. Any test type (unit, integration, smoke, e2e) can import
 * these fixtures by using `import { test } from '../../shared/fixtures'` instead
 * of importing `test` or `it` from 'vitest'.
 *
 * Why this pattern is useful:
 * - Fixtures run ONLY for tests that destructure them (no wasted setup)
 * - Cleanup happens automatically in `finally` blocks (even if tests fail)
 * - Temp dirs are always in os.tmpdir() so they can never accidentally be
 *   your project directory, root, or any important path
 * - Tests that don't need fixtures just use normal `it()` from vitest
 *
 * See: https://vitest.dev/guide/test-context
 * See: https://sdorra.dev/posts/2024-02-12-vitest-tmpdir
 */

import { test as base } from 'vitest';
import { execSync } from 'node:child_process';
import { mkdtempSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';

interface TempGitDirFixture {
  /** Absolute path to a unique temp directory that has been `git init`-ed */
  tempGitDir: string;
}

/**
 * Extended test function that provides a `tempGitDir` fixture.
 *
 * Usage:
 *   import { test } from '../../shared/fixtures';
 *   test('my test', async ({ tempGitDir }) => {
 *     // tempGitDir is a unique /tmp/vitest-git-XXXXXX directory with git init done
 *     // It's automatically cleaned up when the test finishes (pass or fail)
 *   });
 */
export const test = base.extend<TempGitDirFixture>({
  // eslint-disable-next-line no-empty-pattern
  tempGitDir: async ({}, use) => {
    // Create unique temp directory in OS temp (e.g. /tmp/vitest-git-abc123)
    const tempDir = mkdtempSync(join(tmpdir(), 'vitest-git-'));

    // Initialise as a git repo so `git rev-parse --show-toplevel` works from here
    execSync('git init', { cwd: tempDir, stdio: 'ignore' });

    try {
      // Hand the temp dir to the test - test runs during this await
      await use(tempDir);
    } finally {
      // Auto-cleanup after test completes (even on failure)
      // Safe: path is always generated by mkdtempSync under os.tmpdir()
      rmSync(tempDir, { recursive: true, force: true });
    }
  },
});
```
